type CheckpointStore struct {
	// Has unexported fields.
}
    CheckpointStore implements eywa.CheckpointStore using MongoDB, persisting
    reasoning-turn snapshots so an interrupted turn can resume. Each turn is
    one document keyed by turnID; a TTL index on expires_at evicts orphaned
    checkpoints.

func NewCheckpointStore(database *mongodriver.Database) *CheckpointStore
    NewCheckpointStore creates a MongoDB-backed CheckpointStore with the default
    24h checkpoint TTL.

func NewCheckpointStoreWithTTL(database *mongodriver.Database, ttl time.Duration) *CheckpointStore
    NewCheckpointStoreWithTTL creates a MongoDB-backed CheckpointStore with a
    custom TTL for orphan cleanup. A non-positive ttl falls back to the default.

type ChronicleRepository struct {
	// Has unexported fields.
}
    ChronicleRepository persists interaction audit records to MongoDB.

    Immutability note: Chronicle records are append-only by application
    convention — this repository exposes no Update or Delete methods. However,
    MongoDB itself does not enforce write protection; records can be modified
    or deleted by any process with direct database write access. For compliance
    use cases (GDPR, HIPAA, SOC2) that require a legally-binding audit trail,
    supplement with an external append-only log service or enable MongoDB Atlas
    audit log streaming to object storage.

type SpiritRepository struct {
	// Has unexported fields.
}
    Versioned Spirit storage — each Update inserts a new version rather than
    overwriting.

func (r *SpiritRepository) Activate(ctx context.Context, spiritName string, version int) error
    Activate sets only the requested version to active — deactivates all
    other versions of the same Spirit. Both operations run inside a session
    transaction to prevent a window where no version is active.

func (r *SpiritRepository) CountActive(ctx context.Context) (int64, error)

func (r *SpiritRepository) Create(ctx context.Context, spirit *eywa.Spirit) error

func (r *SpiritRepository) Deactivate(ctx context.Context, id string) error

func (r *SpiritRepository) FindActiveByName(ctx context.Context, spiritName string) (*eywa.Spirit, error)

func (r *SpiritRepository) FindActiveByNames(ctx context.Context, spiritNames []string) (map[string]*eywa.Spirit, error)
    FindActiveByNames fetches multiple Spirits in a single query — avoids N+1
    round trips.

func (r *SpiritRepository) FindByID(ctx context.Context, id string) (*eywa.Spirit, error)

func (r *SpiritRepository) FindVersionHistory(ctx context.Context, spiritName string) ([]*eywa.Spirit, error)

func (r *SpiritRepository) GetVersion(ctx context.Context, spiritName string, version int) (*eywa.Spirit, error)

func (r *SpiritRepository) ListActive(ctx context.Context, limit, offset int) ([]*eywa.Spirit, error)

func (r *SpiritRepository) ListAll(ctx context.Context) ([]*eywa.Spirit, error)

func (r *SpiritRepository) RestoreVersion(ctx context.Context, id string) error
    RestoreVersion promotes a specific version to active, deactivating all
    other versions of the same Spirit. Both operations run inside a session
    transaction to prevent a window where no version is active.

func (r *SpiritRepository) Update(ctx context.Context, spiritName string, spirit *eywa.Spirit, changeLog string) (*eywa.Spirit, error)
    Update deactivates the current active version and inserts a new one with an
    incremented version number.
